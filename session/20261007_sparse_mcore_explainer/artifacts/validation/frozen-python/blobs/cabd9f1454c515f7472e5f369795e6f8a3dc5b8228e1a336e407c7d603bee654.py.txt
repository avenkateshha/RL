"""Read-only diagnostics in the previous xToken job's actual cached container."""
import importlib
import importlib.metadata
import inspect
import os
import json
from pathlib import Path
import socket
import subprocess
import sys
import traceback

run = Path(__file__).resolve().parents[1]
report = {"python": sys.executable, "host": socket.gethostname(), "imports": {}, "versions": {}}
for package in ("torch", "ray", "megatron-core", "megatron-bridge", "transformer-engine", "transformers", "nemo-lens"):
    try:
        report["versions"][package] = importlib.metadata.version(package)
    except Exception as exc:
        report["versions"][package] = str(exc)
for module in ("nemo.lens.groups", "nemo.lens", "megatron.core", "megatron.bridge", "nemo_rl.models.policy.workers.megatron_policy_worker", "nemo_rl.algorithms.x_token.loss_utils"):
    try:
        loaded = importlib.import_module(module)
        record = {"status": "PASS", "source": str(loaded.__file__)}
        if "lens" in module:
            record["SpanRegistry"] = hasattr(loaded, "SpanRegistry")
        report["imports"][module] = record
    except Exception:
        report["imports"][module] = {"status": "FAIL", "traceback": traceback.format_exc()}
try:
    import torch
    report["torch_cuda"] = torch.version.cuda
    report["cuda_available"] = torch.cuda.is_available()
    report["gpu_names"] = [torch.cuda.get_device_name(i) for i in range(torch.cuda.device_count())]
    from megatron.bridge import AutoBridge
    report["AutoBridge.from_hf_pretrained"] = str(inspect.signature(AutoBridge.from_hf_pretrained))
    report["AutoBridge.to_megatron_provider"] = str(inspect.signature(AutoBridge.to_megatron_provider))
except Exception:
    report["gpu_bridge_probe_error"] = traceback.format_exc()
report["status"] = "PASS" if all(v["status"] == "PASS" for v in report["imports"].values()) else "FAIL"
name = "runtime-" + Path(sys.prefix).name + "-" + os.environ.get("PREFLIGHT_OVERLAY", "base") + ".json"
(run / name).write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps(report, indent=2), flush=True)
