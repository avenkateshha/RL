from pathlib import Path
from collections import defaultdict
import hashlib
import json
import math
import torch

run = Path('session/20261007_sparse_mcore_explainer/artifacts/validation/runs/R3-dynamic')
reports = [json.loads(path.read_text()) for path in (run / 'controller-reference').glob('*.json')]

reference_groups = defaultdict(list)

    assert report['status'] == 'PASS'
    assert report['native_sparse_teacher_indices'] == [0]
    assert report['native_dense_teacher_indices'] == [1]
    assert not any(report['production_layout_calls'].values())
    metrics = report['reference_metrics']
    ce, kd, ratio = metrics['ce_loss'], metrics['kl_loss'], metrics['kl_loss_scale']
    assert math.isclose(ratio, abs(ce) / abs(kd) if kd else 1.0, rel_tol=1e-6)
    assert math.isclose(metrics['loss'], ce + ratio * kd, rel_tol=1e-6)
    assert math.isclose(metrics['teacher_0/weighted_kl'], 0.3 * metrics['kl_loss_t0'], rel_tol=1e-6)
    assert math.isclose(metrics['teacher_1/weighted_kl'], 0.7 * metrics['kl_loss_t1'], rel_tol=1e-6)

    assert all(math.isclose(ratios[0], value, rel_tol=1e-6) for value in ratios)
